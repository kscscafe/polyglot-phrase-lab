cmake_minimum_required(VERSION 3.16)
project(phrase_audio LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
add_library(silence_detector STATIC src/silence.c)
target_include_directories(silence_detector PUBLIC include)
add_executable(audio_features src/features.cpp)
target_link_libraries(audio_features PRIVATE silence_detector)
enable_testing()
add_executable(audio_test tests/audio_test.cpp)
target_link_libraries(audio_test PRIVATE silence_detector)
add_test(NAME silence_detector_test COMMAND audio_test)
